% figure-macros.tex — shipped geometric figure macros. SOURCE OF TRUTH: this
% file; references/latex-templates.md documents usage. Requires the tikz
% libraries loaded by worksheet-preamble.tex (calc, angles, quotes), so
% \input worksheet-preamble first.
%
% CONTRACT with the layout/prose checkers (do not break it):
%   - Macro names contain rt/tri/fig — that is how tests/check_layout.py and
%     tests/check_prose_consistency.py recognize a macro-style figure.
%   - Mandatory arguments are FLAT braces only (no nested {} inside an arg);
%     the checkers' arg scanners match [^{}]* per group.
%   - Value-bearing args carry exactly the printed label text; scale/styling
%     goes in the single optional [..] argument, which the checkers ignore
%     (so scale=1.2 is never mistaken for a figure value).
%
% Figure conventions (from SKILL.md / latex-templates.md): every number
% printed in a figure must come from the problem's verify JSON; vertices are
% A/B/C with sides a/b/c OPPOSITE them — the same convention verify.py's
% triangle type uses. Right triangles put the right angle at C, so c (= AB)
% is the hypotenuse — the SAME convention scripts/render_figures.py constructs
% (its RIGHT_ANGLE_VERTEX constant): \refrt sits beside renderer-built
% \probfig figures on the page, so the two sources must never disagree.
% tests/test_figure_convention.py enforces this by parsing the
% "% right angle mark" lines below — that trailing comment is load-bearing;
% keep it on every right-angle-mark \draw.
%
% ASCII only: pdflatex (the fallback engine) cannot typeset literal Unicode.

% ── \rtfig[scale]{bottom-leg label}{right-leg label}{hypotenuse label} ──────
% Right triangle, right angle at C (hypotenuse c = AB — the renderer's
% convention). A=(0,0), C=(4,0), B=(4,3): the 3-4-5 shape keeps every label
% clear of the drawn lines at default scale. Bottom leg = b (AC), right leg
% = a (CB), hypotenuse = c (AB) — name sides accordingly in the labels.
% Example: \rtfig{$8$}{$6$}{$x$}   or   \rtfig[0.9]{$b = 8$}{$a = 6$}{$c$}
\newcommand{\rtfig}[4][1.2]{%
  \begin{center}
  \begin{tikzpicture}[scale=#1]
    \coordinate (A) at (0,0);
    \coordinate (B) at (4,3);
    \coordinate (C) at (4,0);
    \draw[thick] (A) -- (C) -- (B) -- cycle;
    \draw (C) ++(-.25,0) -- ++(0,.25) -- ++(.25,0);  % right angle mark
    \node[below left] at (A) {$A$};
    \node[above right] at (B) {$B$};
    \node[below right] at (C) {$C$};
    \node[below] at (2,0) {#2};
    \node[right] at (4,1.5) {#3};
    \node[above left] at (2,1.5) {#4};
  \end{tikzpicture}
  \end{center}}

% ── \trifig[scale]{c}{b}{A-deg}{c-label}{b-label}{a-label}{angle-label} ─────
% General triangle, TO SCALE by construction (SAS): A at the origin, B on the
% x-axis at distance c, C computed from side b and angle A (TikZ trig takes
% degrees). Numeric args #2-#4 are the ACTUAL given values, so the drawing
% cannot disagree with the math; label args #5-#8 carry what is printed.
% Example: \trifig{7}{5}{34}{$c = 7$}{$b = 5$}{$a = ?$}{$34^\circ$}
\newcommand{\trifig}[8][0.6]{%
  \begin{center}
  \begin{tikzpicture}[scale=#1]
    \coordinate (A) at (0,0);
    \coordinate (B) at (#2,0);
    \coordinate (C) at ({#3*cos(#4)},{#3*sin(#4)});
    \draw[thick] (A) -- (B) -- (C) -- cycle;
    \node[below left]  at (A) {$A$};
    \node[below right] at (B) {$B$};
    \node[above]       at (C) {$C$};
    \node[below]       at ($(A)!0.5!(B)$) {#5};
    \node[above left]  at ($(A)!0.5!(C)$) {#6};
    \node[above right] at ($(B)!0.5!(C)$) {#7};
    \pic [draw, angle radius=7mm, angle eccentricity=1.6, "#8"] {angle = B--A--C};
  \end{tikzpicture}
  \end{center}}

% ── \refrt — the value-free reference figure ────────────────────────────────
% SKILL.md's figure-scope rule requires shared labelling conventions to live
% in ONE value-free reference figure placed with the directions; until now no
% template for it existed, so every generation invented one — and an invented
% figure that accidentally carried a number would itself trip the
% all-or-nothing figure rule it exists to satisfy. This macro has zero
% numerals in its output by construction (it takes no arguments and prints
% only letters), and the caption is baked in verbatim so the figure cannot be
% mistaken for a problem's givens. Right angle at C, so sides read: a = BC
% (opposite A), b = AC (opposite B), c = AB (opposite C, the hypotenuse) —
% matching render_figures.py's RIGHT_ANGLE_VERTEX, because this reference
% figure defines the convention students apply to every renderer-built
% figure on the sheet.
\newcommand{\refrt}{%
  \begin{center}
  \begin{tikzpicture}[scale=1.0]
    \coordinate (A) at (0,0);
    \coordinate (B) at (3.6,2.7);
    \coordinate (C) at (3.6,0);
    \draw[thick] (A) -- (C) -- (B) -- cycle;
    \draw (C) ++(-.25,0) -- ++(0,.25) -- ++(.25,0);  % right angle mark
    \node[below left] at (A) {$A$};
    \node[above right] at (B) {$B$};
    \node[below right] at (C) {$C$};
    \node[below] at (1.8,0) {$b$};
    \node[right] at (3.6,1.35) {$a$};
    \node[above left] at (1.8,1.35) {$c$};
    \pic [draw, angle radius=7mm, angle eccentricity=1.6, "$A$"] {angle = C--A--B};
  \end{tikzpicture}
  \end{center}}

% ═══ Segment/parallel marks (CCSS congruence notation) ══════════════════════
% Usable INSIDE any tikzpicture, on named coordinates. Drawn `thick` so the
% marks match primary-edge weight: these are semantics, not auxiliary lines,
% and a hairline mark is a photocopy fade risk (the transversal template's
% hand-plotted chevrons were 0.4pt against 0.8pt edges — inverted hierarchy).
%   \congtick{P}{Q}       one tick at the midpoint of segment PQ
%   \congtickk{P}{Q}      two ticks (second congruence class)
%   \parallelmark{P}{Q}   one chevron pointing P -> Q
%   \parallelmarkk{P}{Q}  two chevrons (second parallel pair)
\newcommand{\congtick}[2]{%
  \draw[thick] ($($(#1)!0.5!(#2)$)!3.2pt!90:(#2)$) -- ($($(#1)!0.5!(#2)$)!3.2pt!270:(#2)$);}
\newcommand{\congtickk}[2]{%
  \draw[thick] ($($($(#1)!0.5!(#2)$)!2.2pt!(#1)$)!3.2pt!90:(#2)$) -- ($($($(#1)!0.5!(#2)$)!2.2pt!(#1)$)!3.2pt!270:(#2)$);
  \draw[thick] ($($($(#1)!0.5!(#2)$)!2.2pt!(#2)$)!3.2pt!90:(#2)$) -- ($($($(#1)!0.5!(#2)$)!2.2pt!(#2)$)!3.2pt!270:(#2)$);}
\newcommand{\parallelmark}[2]{%
  \draw[thick] ($($(#1)!0.5!(#2)$)!5pt!25:(#1)$) -- ($(#1)!0.5!(#2)$) -- ($($(#1)!0.5!(#2)$)!5pt!335:(#1)$);}
\newcommand{\parallelmarkk}[2]{%
  \draw[thick] ($($($(#1)!0.5!(#2)$)!3pt!(#1)$)!5pt!25:(#1)$) -- ($($(#1)!0.5!(#2)$)!3pt!(#1)$) -- ($($($(#1)!0.5!(#2)$)!3pt!(#1)$)!5pt!335:(#1)$);
  \draw[thick] ($($($(#1)!0.5!(#2)$)!3pt!(#2)$)!5pt!25:(#1)$) -- ($($(#1)!0.5!(#2)$)!3pt!(#2)$) -- ($($($(#1)!0.5!(#2)$)!3pt!(#2)$)!5pt!335:(#1)$);}

% ═══ \gridtrifig[scale]{xA}{yA}{xB}{yB}{xC}{yC} — transformation grid ═══════
% CCSS-8 rigid motions/dilations. Fixed -8..8 window with a gridline at EVERY
% integer (the one corpus attempt had lines only every 2 — no line at y = 1
% for a student asked to plot A(-4,1)), labels every 2, gray preimage triangle
% labeled A/B/C for the student to transform. The six numeric args are the
% problem's given vertices — same value-binding contract as \trifig.
\newcommand{\gridtrifig}[7][0.42]{%
  \begin{center}
  \begin{tikzpicture}[scale=#1]
    \draw[gray!35, very thin] (-8,-8) grid (8,8);
    \draw[-{Stealth}] (-8.6,0) -- (8.6,0) node[below right] {$x$};
    \draw[-{Stealth}] (0,-8.6) -- (0,8.6) node[above left] {$y$};
    \foreach \t in {-8,-6,-4,-2,2,4,6,8}{%
      \draw (\t,0.15) -- (\t,-0.15) node[below, font=\scriptsize] {$\t$};
      \draw (0.15,\t) -- (-0.15,\t) node[left, font=\scriptsize] {$\t$};}
    \draw[thick, fill=gray!25] (#2,#3) -- (#4,#5) -- (#6,#7) -- cycle;
    \node[below left, font=\small] at (#2,#3) {$A$};
    \node[below right, font=\small] at (#4,#5) {$B$};
    \node[above, font=\small] at (#6,#7) {$C$};
  \end{tikzpicture}
  \end{center}}

% \tenframefig[cell]{n} — ten frame, 2 rows x 5 cols, first n cells filled
% with black counters, remaining cells empty for the student's pencil.
% Cell default 1.1cm: a first-grader drawing a counter needs >= 1cm.
% Fill order: top row left-to-right, then bottom row (the convention every
% K-1 program teaches).
\newcommand{\tenframefig}[2][1.1]{%
  \begin{tikzpicture}[x=#1cm, y=#1cm, line width=1pt]
    \draw (0,0) rectangle (5,2);
    \foreach \i in {1,...,4} \draw (\i,0) -- (\i,2);
    \draw (0,1) -- (5,1);
    \foreach \k in {1,...,10}{%
      \ifnum\k>#2\relax\else
        \pgfmathtruncatemacro{\row}{\k<=5 ? 1 : 0}%
        \pgfmathtruncatemacro{\colx}{\k<=5 ? \k : \k-5}%
        \fill (\colx-0.5,\row+0.5) circle ({0.33*#1});
      \fi}
  \end{tikzpicture}}

% \fiveframefig[cell]{n} — five frame, one row.
\newcommand{\fiveframefig}[2][1.1]{%

    \foreach \k in {1,...,5}{%
      \ifnum\k>#2\relax\else \fill (\k-0.5,0.5) circle ({0.33*#1}); \fi}
  \end{tikzpicture}}

% \fracbarfig[width-cm]{num}{den} — fraction bar, num of den parts shaded
% with hatching (photocopy-safe). The WHOLE has fixed width (default 9cm) so
% two stacked bars always show the same whole — the property equivalence and
% comparison problems depend on.
\newcommand{\fracbarfig}[3][9]{%
  \begin{tikzpicture}[line width=1pt]
    \pgfmathsetmacro{\w}{#1/#3}
    \foreach \i in {1,...,#3}{\draw ({(\i-1)*\w},0) rectangle ({\i*\w},1.5);}
    \ifnum#2>0
      \foreach \i in {1,...,#2}{\fill[pattern=north east lines]
        ({(\i-1)*\w+0.06},0.06) rectangle ({\i*\w-0.06},1.44);}
    \fi
  \end{tikzpicture}}

% \fraccircfig[radius]{num}{den} — fraction circle, num of den sectors
% hatched. First sector starts at 12 o'clock, filling clockwise (the
% direction fraction-circle manipulatives read).
\newcommand{\fraccircfig}[3][1.7]{%
  \begin{tikzpicture}[line width=1pt]
    \pgfmathsetmacro{\secang}{360/#3}
    \ifnum#2>0
      \foreach \i in {1,...,#2}{%
        \fill[pattern=north east lines]
          (0,0) -- (90-\i*\secang+\secang:#1) arc ({90-\i*\secang+\secang}:{90-\i*\secang}:#1) -- cycle;}
    \fi
    \foreach \i in {1,...,#3}{\draw (0,0) -- (90-\i*\secang:#1);}
    \draw (0,0) circle (#1);
  \end{tikzpicture}}

\newcommand{\arrayfig}[3][0.85]{%

% ---------------------------------------------------------------------------
% \basetenfig{hundreds}{tens}{ones} — base-ten blocks, PROPORTIONAL by
% construction: one = u square, rod = 1 x 10 units, flat = 10 x 10 grid, all
% from the same unit length u = 0.24cm, so ten ones really line up against
% one rod. Grouped hundreds | tens | ones, gaps between groups.
\newlength{\btu}
\newcommand{\basetenfig}[3]{%
  \begin{tikzpicture}[line width=0.9pt]
    \setlength{\btu}{0.24cm}
    \pgfmathsetmacro{\u}{0.24}
    \pgfmathsetmacro{\x}{0}
    % hundreds flats
    \ifnum#1>0
      \foreach \f in {1,...,#1}{%
        \begin{scope}[shift={({(\f-1)*(10*\u+0.35)},0)}]
          \draw[step=\u, line width=0.3pt, gray] (0,0) grid ({10*\u},{10*\u});
          \draw (0,0) rectangle ({10*\u},{10*\u});
        \end{scope}}
    \fi
    \pgfmathsetmacro{\xtens}{#1*(10*\u+0.35)+0.25}
    % tens rods
    \ifnum#2>0
      \foreach \t in {1,...,#2}{%
        \begin{scope}[shift={({\xtens+(\t-1)*(\u+0.3)},0)}]
          \draw (0,0) rectangle (\u,{10*\u});
          \foreach \j in {1,...,9} \draw[line width=0.3pt, gray] (0,{\j*\u}) -- (\u,{\j*\u});
        \end{scope}}
    \fi
    \pgfmathsetmacro{\xones}{\xtens+#2*(\u+0.3)+0.25}
    % ones units
    \ifnum#3>0
      \foreach \o in {1,...,#3}{%
        \pgfmathsetmacro{\ox}{\xones+Mod(\o-1,5)*(\u+0.12)}
        \pgfmathsetmacro{\oy}{(\o<=5 ? 0 : \u+0.12)}
        \draw ({\ox},{\oy}) rectangle ({\ox+\u},{\oy+\u});}
    \fi
  \end{tikzpicture}}
