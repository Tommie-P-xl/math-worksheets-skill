% ── Answer location (constructed, never hand-typed) ─────────────────────────
% A worksheet problem with workspace but no designated answer blank makes the
% grader hunt through scratch work for the final answer, and the student never
% commits to one. The blank is CONSTRUCTED: \problem emits it automatically
% whenever its workspace argument is positive, so generators cannot forget it
% and cannot misplace it. All dimensions live HERE, not at call sites — the
% generated .tex carries no rule lengths for the prose-number checker to
% misread as problem values.
%
%   \ansline   right-aligned "Answer: ____" — emitted by \problem itself;
%              enumerate/\item sheets call it explicitly at the end of each
%              item (tests/check_layout.py enforces one answer-location macro
%              per item there)
%   \ansblank  inline blank for drill formats: "$7 + 5 =$~\ansblank"
%   \noansline opt-out for problems whose worked product IS the answer
%              (graph sketch, proof, construction) — prints nothing, just
%              suppresses the auto-emitted line for the current \problem
%   \answerline{unit}  answer blank with the measurement unit printed after
%              it, e.g. \answerline{ft} — the unit must match the problem's
%              "answer_unit" in the verify JSON (tests/check_answer_line.py
%              enforces the pairing in both directions). Also suppresses the
%              auto \ansline so a unit problem never gets two blanks.
%   \scratchblank  a blank that is NOT an answer: a scaling multiplier, a
%              partial product the student carries forward, a table cell used
%              as working space. Prints exactly like \ansblank and clears the
%              auto-line flag the same way, but tests/check_answer_slots.py
%              does not require a verify entry for it. Using it is a claim —
%              "no answer worth checking lands here" — so do not reach for it
%              to quiet the slot gate on a blank that really does hold an
%              answer. That gate exists because 172 of 300 reviewed cases
%              shipped a printed answer nobody verified.

\newcommand{\noansline}{\global\mws@ansfalse}
% \ansline and \ansblank BOTH clear the flag, which is what makes the auto
% answer line observant instead of unconditional. \problem typesets its stem
% BEFORE testing the flag, so any answer location the stem already contains —
% a per-part \ansline in a nested (a)/(b)/(c) list, or an inline \ansblank in
% a drill — has cleared the flag by the time the test runs, and the parent's
% single trailing line is suppressed. One "Answer: ____" under a problem whose
% three sub-parts each need their own answer is worse than none: it invites the
% student to commit one value for a question that has three.
% Self-clearing is harmless for the parent's own auto-emitted line: it fires
% after the test, and \problem re-arms the flag at the start of every problem.
% Blank length and rule weight grow in an accessibility mode: a student who
% needs 17pt type also needs a blank big enough to write 17pt digits in.
\newcommand{\ansline}{\global\mws@ansfalse\par\nopagebreak\vspace{2pt}\noindent\hfill Answer: \ifmws@bigblanks\rule[-2pt]{5cm}{0.8pt}\else\rule[-1pt]{3.5cm}{0.4pt}\fi}
\newcommand{\ansblank}{\global\mws@ansfalse\ifmws@bigblanks\rule[-2pt]{2.6cm}{0.8pt}\else\rule[-1pt]{1.8cm}{0.4pt}\fi}
% Same rule on the page as \ansblank, deliberately NOT an answer slot. See the
% header note: check_answer_slots.py skips it, so it must only mark working
% space. Defined via \ansblank so the two can never drift apart visually.
\newcommand{\scratchblank}{\ansblank}
\newcommand{\answerline}[1]{\global\mws@ansfalse\par\nopagebreak\vspace{4pt}\hfill\textbf{Answer:}\ \rule[-2pt]{3.5cm}{0.4pt}\ #1}

% \problem[workspace]{stem} — number, stem, and workspace live in ONE minipage
% because LaTeX discards \vspace glue at page/column breaks: workspace left
% outside the box vanishes exactly when a problem lands at a page bottom, and
% on a multi-page sheet that hits several problems per sheet. The minipage is
% unbreakable, so stem + workspace always move to the next page together.
% Default 0pt: answer keys and study guides reuse \problem and must not
% inherit worksheet-sized gaps.
% The answer line is emitted only when workspace > 0pt (so ak_/ss_ documents
% and multi-part parents stay untouched) and only if the stem did not opt out.
% The opt-out flag is reset at the START of each \problem, not the end — a
% \noansline inside an enumerate item must never leak into a later \problem.
\newcommand{\problem}[2][0pt]{\par\ifmws@akmode\vspace{0.5cm plus 0.45cm minus 0.15cm}\else\ifmws@spacious\vspace{0.4cm}\else\vspace{0.15cm}\fi\fi\global\mws@anstrue\noindent\begin{minipage}{\linewidth}\stepcounter{prob}\noindent\textbf{\large\theprob.}\quad #2\par\vspace*{#1}\mws@wsdim=#1\relax\ifdim\mws@wsdim>0pt\ifmws@ans\ansline\fi\fi\end{minipage}\par}
\makeatother

% ── Accessibility modes ─────────────────────────────────────────────────────
% A student whose IEP or 504 plan entitles them to large-print materials could
% not use this tool at all before these existed. Type SIZE must come from the
% document class (article offers only 10/11/12pt), so a large-print sheet opens
%   \documentclass[14pt]{extarticle}   % or 17pt
% and everything below adapts the rest of the page to it.
%
%   \accessiblemode{large}     roomier leading and bigger answer blanks
%   \accessiblemode{dyslexia}  sans-serif text AND math, 1.5 leading, no italics
%   \accessiblemode{both}      both
%
% The dyslexia settings follow the British Dyslexia Association style guide:
% sans-serif faces, generous line spacing, and no italics (italic letterforms
% are the ones most often reported as blurring). sfmath switches MATH to sans
% too, which matters more here than in prose: a worksheet is mostly math, and
% leaving formulas in serif would defeat the point.
% The page budget needs no change: it is computed from content, so a larger
% type size simply produces more pages, which is the honest outcome.
\usepackage{setspace}
\makeatletter
\newcommand{\accessiblemode}[1]{%
  \def\mws@am{#1}%
  \def\mws@amlarge{large}\def\mws@amdys{dyslexia}\def\mws@amboth{both}%
  \ifx\mws@am\mws@amlarge\setstretch{1.25}\global\mws@bigblankstrue\fi
  \ifx\mws@am\mws@amdys\global\mws@sanstrue\setstretch{1.5}\global\mws@bigblankstrue\fi
  \ifx\mws@am\mws@amboth\global\mws@sanstrue\setstretch{1.5}\global\mws@bigblankstrue\fi
  \ifmws@sans
    \renewcommand{\familydefault}{\sfdefault}%
    % sans MATH as well as sans prose: a worksheet is mostly math, so leaving
    % formulas in serif would defeat the purpose of the mode
    \@ifundefined{sfmath@loaded}{\RequirePackage{sfmath}\global\let\sfmath@loaded\relax}{}%
    % italic letterforms are the ones most often reported as blurring, so
    % emphasis becomes bold instead of italic
    \renewcommand{\emph}[1]{\textbf{##1}}%
  \fi}
\makeatother

% ── Locale ──────────────────────────────────────────────────────────────────
% Supporting A4 paper but not the notation that goes with it is half a job. Most
% of the countries that use A4 write the decimal COMMA: a student in Germany or
% France reading "1.5" parses that period as a thousands separator, so this is a
% correctness problem, not a cosmetic one. The multiplication sign differs too
% (\times is taught where \cdot is not, and vice versa).
%
%   \mwslocale{us}   1.5   a \cdot b     (default)
%   \mwslocale{eu}   1,5   a \times b
%
% The verify JSON always stays canonical (period decimals, ASCII operators) so
% the verifier and every gate keep one unambiguous number format. Only the
% PRINTED form is localised, via these macros:
%
%   \dec{3.75}   prints 3.75 or 3,75
%   \mtimes      prints \cdot or \times
%
% Use \dec{} for every decimal in problem text and answer keys when a locale is
% set; a bare 3.75 in the source prints a period regardless and is the one way
% to defeat this.
\makeatletter
\newif\ifmws@eu
\newcommand{\mwslocale}[1]{%
  \def\mws@loc{#1}\def\mws@loceu{eu}%
  \ifx\mws@loc\mws@loceu\global\mws@eutrue\fi}
\newcommand{\dec}[1]{\ifmws@eu\mws@deccomma#1.\@nil\else#1\fi}
\def\mws@deccomma#1.#2\@nil{#1\ifx\relax#2\relax\else{,}\mws@stripdot#2\fi}
\def\mws@stripdot#1.{#1}
\newcommand{\mtimes}{\ifmws@eu\times\else\cdot\fi}
\makeatother

% The key proves the RIGHT answer but says nothing about how to grade a wrong
% one. The verify JSON already declares "traps": the misconception, the
% expression a student following it would evaluate, and the value it produces —
% all machine-checked. Printing that turns grading into teaching: seeing 7.37
% tells the grader the student used cos where tan was needed, which is a
% different conversation from "wrong, minus two".

\newcommand{\commonerror}[3]{\par\noindent{\small #1.~\textit{If they got #2: #3}}\par}

% ── Running headers ─────────────────────────────────────────────────────────
% The running head is ONE line split into two fixed-width boxes: title on the
% left, blanks or a continuation marker on the right. fancyhdr sets [L] and [R]
% at their natural widths, so a title longer than the leftover space simply
% overprinted the Name/Date blanks — and because a running head repeats, the
% collision appeared on EVERY page. Both boxes are now shrink-to-fit within a
% reserved width, so the head can never collide and can never wrap to a second
% line (a wrapped head overprints the body text below it).
% Keeping the title short (<= ~28 chars) is still the right habit: shrink-to-fit
% keeps a long one legible and on one line, it does not make it pretty.
% Ask GEOMETRY for the taller head rather than patching \headheight after the
% fact. geometry computes \topmargin, \headsep and \textheight together; a raw
% \setlength afterwards changes one of them and leaves the rest derived from the
% old value, so the header drifts into the headsep gap and the body sits a few
% points off where every other measurement assumes. \geometry{} re-runs the whole
% computation, which is the only way to keep the page self-consistent.
\geometry{headheight=15pt}
\makeatletter
\newlength{\mws@boxw}
% left-aligned / right-aligned single-line box of exactly the given width
\newcommand{\mws@fitl}[2]{\setlength{\mws@boxw}{#1}\settowidth{\fittedw}{#2}%
  \ifdim\fittedw>\mws@boxw\resizebox{\mws@boxw}{!}{#2}\else\makebox[\mws@boxw][l]{#2}\fi}
\newcommand{\mws@fitr}[2]{\setlength{\mws@boxw}{#1}\settowidth{\fittedw}{#2}%
  \ifdim\fittedw>\mws@boxw\resizebox{\mws@boxw}{!}{#2}\else\makebox[\mws@boxw][r]{#2}\fi}
% title box = whatever the reserved right-hand box leaves, minus a gutter.
% \headwidth is evaluated at shipout, when fancyhdr has finished setting it.
\newcommand{\mws@headtitle}[2]{%
  \mws@fitl{\dimexpr\headwidth-#1-0.5cm\relax}{\small\bfseries #2}}

% \akheader{Short Topic} — answer key
% Also switches \ans{...} to the COMPACT boxed form: check_answer_key.py binds
% printed answers through the \ans/\boxed macro names, so a space-saving layout
% must keep the macro name and change only what it typesets — a new macro
% (\aksol etc.) would be invisible to the gate. The TeXbook end-of-proof idiom
% below puts the \fbox on the same line as the last worked step when it fits
% and flush-right on the next line when it does not; a bare \hfill\fbox would
% instead produce Overfull \hbox on long answers, which scripts/check_log.py
% rightly turns into build failures. Study guides keep the \boldsymbol form
% (\ssheader never redefines \ans).
\newcommand{\akheader}[1]{%
  \pagestyle{fancy}\fancyhf{}%
  % The right slot carries nothing: "For instructor/parent use" restated what
  % "Answer Key" already says, and every centimetre it held was width the title
  % could not use — which is what forced the title to shrink. Reserve 0cm and
  % the title gets the whole head. Rule: the header's right slot is for
  % information the reader must ACT on (the Name/Date blanks). Labels that only
  % restate the document type belong nowhere.
  \fancyhead[L]{\mws@headtitle{0cm}{#1 --- Answer Key}}%
  \fancyfoot[C]{\thepage}%
  \fancyfoot[L]{\small\schoolname}%
  \renewcommand{\headrulewidth}{0.4pt}%
  \mws@akmodetrue
  \raggedbottom
  \setlength{\parskip}{2pt plus 1.5pt}%
  % Keep the compact box valid in both authoring forms. In prose, retain the
  % flush-right end-of-proof layout; inside `$...$`, stay in math mode instead
  % of emitting paragraph glue (or a second pair of math delimiters).
  \renewcommand{\ans}[1]{\ifmmode\boxed{##1}\else{\unskip\nobreak\hfil\penalty50\hskip1em\null\nobreak\hfil\mbox{\fbox{\ensuremath{\,##1\,}}}\parfillskip=0pt\finalhyphendemerits=0\par}\fi}}

% \ans{...} — the printed final answer of a worked example. check_answer_key.py
% binds study guides through this macro (and \boxed); a result set in bare
% \boldsymbol is invisible to the gate, so every worked example ends in \ans.
%
% \ensuremath, not a bare \boldsymbol, because \boldsymbol is math-only and this
% definition is what a STUDY GUIDE gets: \akheader replaces \ans with a
% text-safe compact box, \ssheader does not, so in an ss_ document \ans has
% always been the raw \boldsymbol. Every shipped exemplar writes it inside
% $...$, so that held right up until someone followed SKILL.md's prose ("print
% each result with \ans{...}") instead of copying a template — and got
% "! Missing $ inserted", a fatal error naming the symptom two lines away from
% the cause, on the last gate of the chain. \ensuremath is a no-op in math mode,
% so every existing study guide renders byte-identically; it only adds the
% missing text-mode path. Do not "simplify" it back.
\newcommand{\ans}[1]{\ensuremath{\boldsymbol{#1}}}

% ═══ Worksheet graph & chart house styles ═══════════════════════════════════
% One set of styles every figure on every sheet inherits, so a document's
% graphs match each other and survive a photocopier. The rules they encode,
% each measured against rendered corpus pages before shipping:
%   * black ink; semantics carried by dash pattern, fill pattern, line weight
%     and LABELS — never by hue (a red-vs-blue distinction is two identical
%     dots after one photocopy, and a shipped grade-3 comparison sheet was
%     unanswerable in grayscale for exactly that reason)
%   * line-weight hierarchy: minor grid 0.25pt < major grid 0.4pt
%     < axis 0.8pt < curve/data 1.1pt — the data must be the darkest thing
%   * fixed PHYSICAL units (x=7mm), never width/height stretching: unit
%     squares stay pencil-sized and slopes render true (a corpus piecewise
%     graph drew slope 1 at ~30 degrees from independent width/height)
%   * Stealth arrowheads; tick labels white-backed so a curve can never
%     strike a number ("2pi", "-1" and "6" were all struck in corpus renders)
\pgfplotsset{
  % Draw order for graphs: grid at the bottom, then axis lines, then the
  % curves, then white-backed tick labels, then descriptions. This custom
  % layer set — NOT `axis on top` — is load-bearing: with `axis on top` a
  % dashed asymptote lying on an integer gridline (x = 2) is painted over by
  % the grid and disappears. Verified by rendering both.
  layers/wsgraph/.define layer set=
    {axis background, axis grid, axis ticks, axis lines, main,
     axis tick labels, axis descriptions, axis foreground}
    {grid style={/pgfplots/on layer=axis grid},
     tick style={/pgfplots/on layer=axis ticks},
     axis line style={/pgfplots/on layer=axis lines},
     label style={/pgfplots/on layer=axis descriptions},
     legend style={/pgfplots/on layer=axis descriptions},
     title style={/pgfplots/on layer=axis descriptions},
     ticklabel style={/pgfplots/on layer=axis tick labels},
     axis background@ style={/pgfplots/on layer=axis background}},
  wsaxisbase/.style={
    set layers=wsgraph,
    axis lines=center,
    axis line style={line width=0.8pt,
      {Stealth[length=2.4mm, width=2.2mm]}-{Stealth[length=2.4mm, width=2.2mm]},
      shorten <=-3.5mm, shorten >=-3.5mm},
    xlabel={$x$}, ylabel={$y$},
    xlabel style={anchor=west, font=\small, xshift=3.5mm},
    ylabel style={anchor=south, font=\small, yshift=3.5mm},
    grid=both,
    minor grid style={line width=0.25pt, draw=gray!35},
    major grid style={line width=0.4pt, draw=gray!55},
    tick style={draw=none},
    tick label style={font=\small, fill=white, inner sep=1pt},
    xticklabel={\ifdim\tick pt=0pt\else$\pgfmathprintnumber{\tick}$\fi},
    yticklabel={\ifdim\tick pt=0pt\else$\pgfmathprintnumber{\tick}$\fi},
    execute at end axis={\begin{pgfonlayer}{axis tick labels}
      \node[below left, font=\scriptsize, inner sep=2pt, fill=white]
        at (axis cs:0,0) {$O$};\end{pgfonlayer}},
    enlargelimits=false, clip=false},
  % Student plotting grid: 7mm unit squares, a gridline and label at EVERY
  % integer. The grid this replaces had lines only every 2 units — no line
  % at any odd integer, on the figure whose whole job is plotting (3,-1).
  wsgrid/.style={wsaxisbase, x=7mm, y=7mm, xtick distance=1, ytick distance=1},
  % Wide range (to +-10): 5.5mm units, labels every 2, a line still every 1.
  wsgridwide/.style={wsaxisbase, x=5.5mm, y=5.5mm,
    xtick distance=2, ytick distance=2, minor tick num=1,
    tick label style={font=\scriptsize, fill=white, inner sep=1pt}},
  % Tall-range sketching (cubics): y compressed 2:1, still gridded.
  wsfuntall/.style={wsaxisbase, x=7mm, y=3.5mm,
    xtick distance=1, ytick distance=4, minor y tick num=1},
  % Trig: give width/height and pi-multiple xtick/xticklabels per figure.
  wstrig/.style={wsaxisbase, ytick distance=1, minor y tick num=1},
  % Quadrant-I context grid (rates, proportions): axes start AT the origin,
  % arrows outward only, unit words go at the arrow tips, never inside the
  % plot area.
  wsgridq1/.style={wsaxisbase, xmin=0, ymin=0,
    axis line style={line width=0.8pt,
      -{Stealth[length=2.4mm, width=2.2mm]}, shorten >=-3.5mm},
    execute at end axis={\begin{pgfonlayer}{axis tick labels}
      \node[below left, font=\scriptsize, inner sep=2pt]
        at (axis cs:0,0) {$0$};\end{pgfonlayer}},
    x=7mm, y=7mm, xtick distance=1, ytick distance=1},
  % ── data displays ──
  % wsbar: single-series bar chart a student READS VALUES OFF. No printed
  % value labels: horizontal gridlines at a countable step do that work.
  % Call-site rules: pick the ytick step from {1,2,5,10} so every bar top
  % lands ON a gridline; set ymax one step above the tallest bar.
  % NOTE the trailing comma inside cycle list={{...},} — a single-entry cycle
  % list without it is split into per-token entries by pgfkeys brace
  % stripping, and the style silently falls back to the blue default.
  wsbar/.style={
    ybar, axis lines=left, ymin=0,
    bar width=22pt,
    enlarge x limits={abs=0.75cm},
    ymajorgrids=true,
    major grid style={line width=0.4pt, draw=gray!60},
    axis line style={line width=0.6pt},
    every tick/.append style={black, line width=0.5pt},
    tick label style={font=\small},
    label style={font=\small},
    cycle list={{fill=gray!30, draw=black, line width=0.6pt},},
    height=5.6cm, width=10.5cm},
  % wsbar labeled: printed values ON the bars — ONLY for charts feeding
  % computation (means, fraction-of-total). Never for a read_data sheet whose
  % query is value/max_key/min_key/difference: the label answers the question
  % for the student, which is the defect a judge found shipped on curr-200.
  wsbar labeled/.style={
    wsbar,
    nodes near coords,
    every node near coord/.append style={font=\small, color=black}},
  % wsline: line chart over ordered categories/time; same read-off-the-grid
  % contract as wsbar.
  wsline/.style={
    axis lines=left, ymin=0,
    enlarge x limits={abs=0.45cm},
    ymajorgrids=true,
    major grid style={line width=0.4pt, draw=gray!60},
    axis line style={line width=0.6pt},
    tick label style={font=\small},
    label style={font=\small},
    cycle list={{black, thick, mark=*, mark size=2.2pt},},
    height=5.6cm, width=10.5cm},
  % wshist: a TRUE histogram — touching bars, ticks and labels at BIN EDGES
  % (feed n+1 coordinates: each bin's left edge plus the closing right edge;
  % the last y is ignored). A gapped ybar chart with interval labels is a
  % bar chart of intervals, not a histogram — shape-of-distribution language
  % (skew, tail) depends on the continuity the gaps break.
  wshist/.style={
    ybar interval, axis lines=left, ymin=0,
    xtick=data,
    x tick label as interval=false,
    xmajorgrids=false, enlarge x limits=false,
    ymajorgrids=true,
    major grid style={line width=0.4pt, draw=gray!60},
    axis line style={line width=0.6pt},
    tick label style={font=\small},
    label style={font=\small},
    cycle list={{fill=gray!25, draw=black, line width=0.6pt},},
    height=5.6cm, width=11cm},
  % wsboxplot: horizontal box plot over a numeric axis. No y-axis (a box plot
  % has no y meaning). Feed boxplot prepared the SAME five-number summary the
  % stats entries verify — never raw data: pgfplots quartiles by an
  % interpolating convention and verify.py uses school median-of-halves, and
  % on [4,6,7,9,11,12,18] they disagree (median 9 vs 8).
  wsboxplot/.style={
    axis x line=bottom,
    hide y axis,
    ymin=0.5, ymax=1.6,
    height=3.4cm, width=10.5cm,
    tick label style={font=\small},
    label style={font=\small},
    axis line style={line width=0.6pt},
    xmajorgrids=true,
    major grid style={line width=0.3pt, draw=gray!45},
    every axis plot/.append style={black, line width=0.8pt, fill=gray!15},
    boxplot/draw direction=x,
    boxplot/box extend=0.55,
    boxplot/whisker extend=0.35},
  % wsboxplot pair: two summaries on ONE shared axis for compare items —
  % one axis cannot mis-scale, which the two-panel groupplot template did.
  % Give each \addplot boxplot prepared={draw position=1 / 2, ...} and name
  % the rows with yticklabels.
  wsboxplot pair/.style={
    wsboxplot,
    hide y axis=false,
    y axis line style={draw=none},
    ytick style={draw=none},
    ymin=0.4, ymax=2.6,
    height=4.6cm,
    ytick={1,2}},
}
